/**
 * Historique Envers pour N'IMPORTE QUELLE entité @Audited.
 * Aucune classe à créer par entité : le snapshot est lu via le métamodèle JPA.
 *
 *  - une relation vers une entité devient "<nom>Id"
 *  - une collection devient une liste triée (identifiants ou valeurs)
 *  - un champ @NotAudited (ex. mot de passe) n'apparaît jamais
 */
@Service
@Transactional(readOnly = true)
public class AuditQuery {

    private static final Set<String> IGNORED = Set.of("id", "version");

    @PersistenceContext
    private EntityManager entityManager;

    private final AuditRegistry registry;

    public AuditQuery(AuditRegistry registry) {
        this.registry = registry;
    }

    @SuppressWarnings("unchecked")
    public List<AuditResponse> getHistory(String entityKey, Long id) {
        Class<?> entityClass = registry.resolve(entityKey);

        AuditReader reader = AuditReaderFactory.get(entityManager);

        List<Object[]> revisions = reader.createQuery()
                .forRevisionsOfEntity(entityClass, false, true)
                .add(AuditEntity.id().eq(id))
                .addOrder(AuditEntity.revisionNumber().asc())
                .getResultList();

        return revisions.stream()
                .map(row -> {
                    RevisionInfo revision = (RevisionInfo) row[1];
                    RevisionType type = (RevisionType) row[2];

                    return new AuditResponse(
                            id,
                            revision.getId(),
                            type.name(),
                            Instant.ofEpochMilli(revision.getTimestamp()),
                            revision.getUsername(),
                            snapshot(row[0], entityClass)
                    );
                })
                .toList();
    }

    private Map<String, Object> snapshot(Object entity, Class<?> entityClass) {
        Map<String, Object> data = new LinkedHashMap<>();

        EntityType<?> type = entityManager.getMetamodel().entity(entityClass);

        for (Attribute<?, ?> attribute : type.getSingularAttributes()) {

            String name = attribute.getName();

            if (IGNORED.contains(name) || isNotAudited(attribute)) {
                continue;
            }

            Object value = entity == null ? null : read(entity, attribute);

            if (value instanceof BaseEntity reference) {
                data.put(name + "Id", reference.getId());
            } else {
                data.put(name, value);
            }
        }

        for (PluralAttribute<?, ?, ?> attribute : type.getPluralAttributes()) {

            if (isNotAudited(attribute)) {
                continue;
            }

            Object value = entity == null ? null : read(entity, attribute);

            data.put(attribute.getName(), describeCollection(value));
        }

        return data;
    }

    private boolean isNotAudited(Attribute<?, ?> attribute) {
        Member member = attribute.getJavaMember();

        return member instanceof AnnotatedElement element
                && element.isAnnotationPresent(NotAudited.class);
    }

    private Object read(Object entity, Attribute<?, ?> attribute) {
        Member member = attribute.getJavaMember();
        try {
            if (member instanceof Field field) {
                field.setAccessible(true);
                return field.get(entity);
            }
            if (member instanceof Method method) {
                method.setAccessible(true);
                return method.invoke(entity);
            }
        } catch (ReflectiveOperationException e) {
            throw new IllegalStateException(
                    "Lecture impossible de " + attribute.getName(), e);
        }
        return null;
    }

    // Collection -> liste triée (identifiants pour des entités, texte sinon).
    private Object describeCollection(Object value) {

        if (!(value instanceof Collection<?> collection)) {
            return null;
        }

        try {
            List<Object> items = new ArrayList<>();

            for (Object element : collection) {
                items.add(element instanceof BaseEntity reference
                        ? reference.getId()
                        : String.valueOf(element));
            }

            items.sort((a, b) -> {
                if (a instanceof Number x && b instanceof Number y) {
                    return Long.compare(x.longValue(), y.longValue());
                }
                return String.valueOf(a).compareTo(String.valueOf(b));
            });

            return items;

        } catch (RuntimeException e) {
            // Collection non chargeable (ex. entité supprimée) : on n'affiche rien.
            return null;
        }
    }
}

package com.kacheche.audit;

import com.kacheche.exception.ResourceNotFoundException;
import jakarta.persistence.EntityManagerFactory;
import jakarta.persistence.Table;
import jakarta.persistence.metamodel.EntityType;
import org.hibernate.envers.Audited;
import org.springframework.stereotype.Component;

import java.util.Map;
import java.util.TreeMap;

/**
 * Découvre automatiquement toutes les entités @Audited au démarrage.
 * Clé d'accès = nom de table (@Table), ex : "departments", "budgets", "expenses".
 * Une nouvelle entité @Audited + @Table est auditable sans écrire une ligne de plus.
 */
@Component
public class AuditRegistry {

    private final Map<String, Class<?>> entities = new TreeMap<>();

    public AuditRegistry(EntityManagerFactory entityManagerFactory) {
        for (EntityType<?> type : entityManagerFactory.getMetamodel().getEntities()) {
            Class<?> entityClass = type.getJavaType();

            if (!entityClass.isAnnotationPresent(Audited.class)) {
                continue;
            }

            Table table = entityClass.getAnnotation(Table.class);
            String key = table != null && !table.name().isBlank()
                    ? table.name()
                    : entityClass.getSimpleName().toLowerCase() + "s";

            entities.put(key, entityClass);
        }
    }

    public Class<?> resolve(String key) {
        Class<?> entityClass = entities.get(key);
        if (entityClass == null) {
            throw new ResourceNotFoundException(
                    "Entité auditée inconnue : " + key + ". Disponibles : " + entities.keySet());
        }
        return entityClass;
    }
}

/**
 * Sécurité de l'API : authentification JWT sans session + autorisation RBAC.
 * Chaque route exige une PERMISSION (jamais un nom de rôle) : les rôles ne sont
 * que des ensembles de permissions, modifiables en base sans toucher au code.
 */
@Configuration
@EnableMethodSecurity
public class SecurityConfig {

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    // Utilise automatiquement AppUserDetailsService + PasswordEncoder.
    @Bean
    public AuthenticationManager authenticationManager(
            AuthenticationConfiguration configuration) throws Exception {
        return configuration.getAuthenticationManager();
    }

    @Bean
    public SecurityFilterChain securityFilterChain(
            HttpSecurity http,
            JwtService jwtService,
            AppUserDetailsService userDetailsService,
            RestAuthenticationEntryPoint authenticationEntryPoint,
            RestAccessDeniedHandler accessDeniedHandler) throws Exception {

        JwtAuthenticationFilter jwtFilter =
                new JwtAuthenticationFilter(jwtService, userDetailsService);

        return http
                .cors(Customizer.withDefaults())
                .csrf(AbstractHttpConfigurer::disable)
                .httpBasic(AbstractHttpConfigurer::disable)
                .formLogin(AbstractHttpConfigurer::disable)
                .logout(AbstractHttpConfigurer::disable)

                .sessionManagement(session ->
                        session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))

                .exceptionHandling(handling -> handling
                        .authenticationEntryPoint(authenticationEntryPoint)
                        .accessDeniedHandler(accessDeniedHandler))

                .authorizeHttpRequests(auth -> {

                    // ---- Routes publiques ----
                    auth.requestMatchers(HttpMethod.OPTIONS, "/**").permitAll();
                    auth.requestMatchers("/error").permitAll();
                    auth.requestMatchers(HttpMethod.POST,
                            "/api/auth/login",
                            "/api/auth/refresh",
                            "/api/auth/logout").permitAll();

                    // ---- Utilisateur connecté (me, change-password) ----
                    auth.requestMatchers("/api/auth/**").authenticated();

                    // ---- Permissions "métier" ----
                    auth.requestMatchers("/api/dashboard/**")
                            .hasAuthority(Permissions.DASHBOARD_READ);

                    auth.requestMatchers("/api/audit/**")
                            .hasAuthority(Permissions.AUDIT_READ);

                    auth.requestMatchers(HttpMethod.PATCH, "/api/expenses/*/approve")
                            .hasAuthority(Permissions.EXPENSE_APPROVE);

                    auth.requestMatchers(HttpMethod.PATCH, "/api/expenses/*/reject")
                            .hasAuthority(Permissions.EXPENSE_REJECT);

                    auth.requestMatchers(HttpMethod.POST, "/api/users/*/reset-password")
                            .hasAuthority(Permissions.USER_RESET_PASSWORD);

                    // ---- Écrans de gestion des utilisateurs : listes déroulantes ----
                    auth.requestMatchers(HttpMethod.GET, "/api/roles/**")
                            .hasAnyAuthority(
                                    Resource.ROLE.permission(Action.READ),
                                    Resource.USER.permission(Action.CREATE),
                                    Resource.USER.permission(Action.UPDATE));

                    auth.requestMatchers(HttpMethod.GET, "/api/permissions/**")
                            .hasAnyAuthority(
                                    Resource.ROLE.permission(Action.READ),
                                    Resource.ROLE.permission(Action.CREATE),
                                    Resource.ROLE.permission(Action.UPDATE));

                    // ---- CRUD générique : une règle par ressource et par action ----
                    for (Resource resource : Resource.values()) {
                        for (Action action : Action.values()) {
                            auth.requestMatchers(action.httpMethod(), resource.apiPattern())
                                    .hasAuthority(resource.permission(action));
                        }
                    }

                    // ---- Refus par défaut ----
                    // Toute route d'API non déclarée ci-dessus est interdite.
                    auth.requestMatchers("/api/**").denyAll();

                    auth.anyRequest().authenticated();
                })

                .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class)
                .build();
    }

    @Bean
    public CorsConfigurationSource corsConfigurationSource() {

        CorsConfiguration configuration = new CorsConfiguration();

        configuration.setAllowedOriginPatterns(
                List.of(
                        "http://localhost:[*]",
                        "http://127.0.0.1:[*]"
                )
        );

        configuration.setAllowedMethods(
                List.of("GET", "POST", "PUT", "DELETE", "PATCH", "OPTIONS")
        );

        configuration.setAllowedHeaders(List.of("*"));

        configuration.setAllowCredentials(true);

        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", configuration);

        return source;
    }
}

/**
 * Contrôleur CQRS générique.
 * Les écritures passent par la Command, les lectures par la Query.
 * E = entité, Q = request (commande), S = response, QS = Query, CS = Command.
 */
@CrossOrigin(
        origins = {"http://localhost:4200", "http://127.0.0.1:4200"},
        allowCredentials = "true"
)
public abstract class BaseController<E extends BaseEntity, Q extends VersionedRequest, S,
        QS extends BaseQuery<E, S, ?>, CS extends BaseCommand<E, Q, ?>> {

    protected final QS query;
    protected final CS command;

    protected BaseController(final QS query, final CS command) {
        this.query = query;
        this.command = command;
    }

    @GetMapping
    public ResponseEntity<List<S>> findAll() {
        return ResponseEntity.ok(query.findAll());
    }

    @GetMapping("/{id}")
    public ResponseEntity<S> findById(@PathVariable Long id) {
        return ResponseEntity.ok(query.findById(id));
    }

    @PostMapping
    public ResponseEntity<S> create(@Valid @RequestBody Q request) {
        Long id = command.create(request);
        return ResponseEntity.status(HttpStatus.CREATED).body(query.findById(id));
    }

    @PutMapping("/{id}")
    public ResponseEntity<S> update(@PathVariable Long id, @Valid @RequestBody Q request) {
        command.update(id, request);
        return ResponseEntity.ok(query.findById(id));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> delete(@PathVariable Long id) {
        command.delete(id);
        return ResponseEntity.noContent().build();
    }
}

import java.util.stream.Collectors;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final String DEFAULT_BAD_CREDENTIALS = "Bad credentials";

    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<ErrorResponse> handleNotFound(
            ResourceNotFoundException exception, HttpServletRequest request) {
        return build(HttpStatus.NOT_FOUND, "Not Found", exception.getMessage(), request);
    }

    @ExceptionHandler(DuplicateResourceException.class)
    public ResponseEntity<ErrorResponse> handleDuplicate(
            DuplicateResourceException exception, HttpServletRequest request) {
        return build(HttpStatus.CONFLICT, "Conflict", exception.getMessage(), request);
    }

    // Optimistic locking : la ressource a été modifiée par quelqu'un d'autre
    @ExceptionHandler(OptimisticLockingFailureException.class)
    public ResponseEntity<ErrorResponse> handleOptimisticLock(
            OptimisticLockingFailureException exception, HttpServletRequest request) {
        return build(HttpStatus.CONFLICT, "Version Conflict",
                "Cette ressource a été modifiée par un autre utilisateur. "
                        + "Rechargez-la puis recommencez.", request);
    }

    // Transition ou opération interdite (ex : supprimer son propre compte)
    @ExceptionHandler(IllegalStateException.class)
    public ResponseEntity<ErrorResponse> handleIllegalState(
            IllegalStateException exception, HttpServletRequest request) {
        return build(HttpStatus.CONFLICT, "Conflict", exception.getMessage(), request);
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<ErrorResponse> handleBadRequest(
            IllegalArgumentException exception, HttpServletRequest request) {
        return build(HttpStatus.BAD_REQUEST, "Bad Request", exception.getMessage(), request);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorResponse> handleValidation(
            MethodArgumentNotValidException exception, HttpServletRequest request) {

        String message = exception.getBindingResult()
                .getFieldErrors()
                .stream()
                .map(DefaultMessageSourceResolvable::getDefaultMessage)
                .collect(Collectors.joining("; "));

        return build(HttpStatus.BAD_REQUEST, "Validation Failed", message, request);
    }

    // ---------- SÉCURITÉ ----------

    // Identifiants invalides, ou jeton de rafraîchissement invalide
    @ExceptionHandler(BadCredentialsException.class)
    public ResponseEntity<ErrorResponse> handleBadCredentials(
            BadCredentialsException exception, HttpServletRequest request) {

        String message = DEFAULT_BAD_CREDENTIALS.equals(exception.getMessage())
                ? "Nom d’utilisateur ou mot de passe incorrect."
                : exception.getMessage();

        return build(HttpStatus.UNAUTHORIZED, "Unauthorized", message, request);
    }

    // Trop de tentatives échouées
    @ExceptionHandler(LockedException.class)
    public ResponseEntity<ErrorResponse> handleLocked(

    }

        return build(HttpStatus.FORBIDDEN, "Forbidden",
                "Vous n’avez pas les permissions nécessaires pour cette action.", request);
    }

    }

    }

    }

    public User toEntity(UserRequest request) {

        User user = new User();

        user.setUsername(request.username().trim().toLowerCase(Locale.ROOT));
        user.setFullName(blankToNull(request.fullName()));
        user.setEmail(blankToNull(request.email()));
        user.setEnabled(request.enabled() == null || request.enabled());

        // Mot de passe haché dès ici : le clair ne circule pas plus loin.

            user.setPassword(passwordEncoder.encode(request.password()));
        }

        // Un nouveau compte doit changer son mot de passe provisoire.
        user.setMustChangePassword(true);

        // Rôles : simples références, résolues par UserCommand.prepare().
        Set<Role> roles = request.roleIds().stream()
                .map(Role::new)
                .collect(Collectors.toCollection(HashSet::new));

        user.setRoles(roles);

        return user;
    }

        return new UserResponse(

                user.getFullName(),
                user.getEmail(),

                user.isLocked(),
                user.isMustChangePassword(),
                user.getLastLoginAt(),
                user.getRoles().stream()
                        .sorted(Comparator.comparing(Role::getName))
                        .map(role -> new RoleSummaryResponse(role.getId(), role.getName()))
                        .toList(),
                user.getVersion()
        );

        this.authenticationManager = authenticationManager;
        this.jwtService = jwtService;
        this.refreshTokenService = refreshTokenService;
        this.loginAttemptService = loginAttemptService;
        this.userRepository = userRepository;
        this.passwordEncoder = passwordEncoder;
    }

    // ---------- CONNEXION ----------
    public AuthResponse login(LoginRequest request) {

        String username = request.username().trim().toLowerCase(Locale.ROOT);

        Authentication authentication;

        try {
            // Vérifie le mot de passe, l'état activé et le verrouillage du compte.
            authentication = authenticationManager.authenticate(
                    UsernamePasswordAuthenticationToken.unauthenticated(
                            username, request.password()));

        } catch (BadCredentialsException e) {
            loginAttemptService.recordFailure(username);
            throw e;
        }

        AppUserDetails user = (AppUserDetails) authentication.getPrincipal();

        loginAttemptService.recordSuccess(user.getId());

        return buildResponse(user, refreshTokenService.issue(user.getId()));
    }

    // ---------- RAFRAÎCHISSEMENT (rotation du jeton) ----------
    @Transactional(noRollbackFor = BadCredentialsException.class)
    public AuthResponse refresh(String rawRefreshToken) {

        RefreshTokenService.Rotation rotation = refreshTokenService.rotate(rawRefreshToken);

        return buildResponse(
                AppUserDetails.from(rotation.user()),
                rotation.refreshToken());
    }

    // ---------- DÉCONNEXION ----------
    public void logout(String rawRefreshToken) {
        if (rawRefreshToken != null && !rawRefreshToken.isBlank()) {
            refreshTokenService.revoke(rawRefreshToken);
        }
    }

    // ---------- CHANGEMENT DE MOT DE PASSE ----------
    @Transactional
    public AuthResponse changePassword(Long userId, ChangePasswordRequest request) {

        User user = userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException("Utilisateur introuvable."));

        if (!passwordEncoder.matches(request.currentPassword(), user.getPassword())) {
            throw new IllegalArgumentException("Le mot de passe actuel est incorrect.");
        }

        if (passwordEncoder.matches(request.newPassword(), user.getPassword())) {
            throw new IllegalArgumentException(
                    "Le nouveau mot de passe doit être différent de l’ancien.");
        }

        user.setPassword(passwordEncoder.encode(request.newPassword()));
        user.setMustChangePassword(false);
        userRepository.saveAndFlush(user);

        // Toutes les anciennes sessions sont coupées ; on en ouvre une nouvelle.
        refreshTokenService.revokeAllForUser(userId);

        return buildResponse(
                AppUserDetails.from(user),
                refreshTokenService.issue(userId));
    }

    private AuthResponse buildResponse(AppUserDetails user, String refreshToken) {
        return new AuthResponse(
                jwtService.generateAccessToken(user),
                refreshToken,
                TOKEN_TYPE,
                jwtService.getAccessTokenSeconds(),
                CurrentUserResponse.from(user)
        );
    }

 * Les droits sont relus en base : un compte désactivé, verrouillé ou dont les
 * rôles ont changé est pris en compte immédiatement.

public class JwtAuthenticationFilter extends OncePerRequestFilter {

    public static final String ERROR_ATTRIBUTE = "kacheche.auth.error";

    private static final String BEARER_PREFIX = "Bearer ";

    // Seules routes autorisées tant que le mot de passe provisoire n'est pas changé
    private static final Set<String> PASSWORD_CHANGE_ALLOWED = Set.of(
            "/api/auth/me",
            "/api/auth/change-password",
            "/api/auth/logout"
    );

    private final JwtService jwtService;
    private final AppUserDetailsService userDetailsService;

    public JwtAuthenticationFilter(
            JwtService jwtService,
            AppUserDetailsService userDetailsService) {

        this.jwtService = jwtService;
        this.userDetailsService = userDetailsService;
    }

    @Override
    protected void doFilterInternal(
            HttpServletRequest request,
            HttpServletResponse response,
            FilterChain filterChain) throws ServletException, IOException {

        String header = request.getHeader(HttpHeaders.AUTHORIZATION);

        if (header != null && header.startsWith(BEARER_PREFIX)) {

            try {
                String username = jwtService.parseAccessToken(
                        header.substring(BEARER_PREFIX.length()).trim());

                AppUserDetails user =
                        (AppUserDetails) userDetailsService.loadUserByUsername(username);

                if (user.isEnabled() && user.isAccountNonLocked()) {

                    if (user.isMustChangePassword()
                            && !PASSWORD_CHANGE_ALLOWED.contains(request.getRequestURI())) {

                        RestErrors.write(request, response, 403, "Forbidden",
                                "Vous devez changer votre mot de passe avant de continuer.",
                                "PASSWORD_CHANGE_REQUIRED");
                        return;
                    }

                    UsernamePasswordAuthenticationToken authentication =
                            UsernamePasswordAuthenticationToken.authenticated(
                                    user, null, user.getAuthorities());

                    SecurityContextHolder.getContext().setAuthentication(authentication);

                } else {
                    request.setAttribute(ERROR_ATTRIBUTE, "TOKEN_INVALID");
                }

            } catch (ExpiredJwtException e) {
                request.setAttribute(ERROR_ATTRIBUTE, "TOKEN_EXPIRED");

            } catch (JwtException | IllegalArgumentException | UsernameNotFoundException e) {
                request.setAttribute(ERROR_ATTRIBUTE, "TOKEN_INVALID");
            }
        }

        filterChain.doFilter(request, response);
    }
}

package com.kacheche.security;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.Date;
import java.util.UUID;

/**
 * Émission et vérification des jetons d'accès JWT (signés HMAC).
 * Le jeton ne contient que l'identité : les rôles et permissions sont relus
 * en base à chaque requête, donc toute modification de droits est immédiate.
 */
@Service
public class JwtService {

    private static final String ISSUER = "kacheche";
    private static final String TYPE_CLAIM = "typ";
    private static final String ACCESS_TYPE = "access";

    private final SecretKey key;
    private final long accessTokenSeconds;

    public JwtService(
            @Value("${app.security.jwt.secret}") String secret,
            @Value("${app.security.jwt.access-token-minutes:30}") long accessTokenMinutes) {

        if (secret == null || secret.getBytes(StandardCharsets.UTF_8).length < 32) {
            throw new IllegalStateException(
                    "app.security.jwt.secret doit contenir au moins 32 caractères.");
        }

        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
        this.accessTokenSeconds = accessTokenMinutes * 60;
    }

    public String generateAccessToken(AppUserDetails user) {

        Instant now = Instant.now();

        return Jwts.builder()
                .id(UUID.randomUUID().toString())
                .issuer(ISSUER)
                .subject(user.getUsername())
                .claim(TYPE_CLAIM, ACCESS_TYPE)
                .claim("uid", user.getId())
                .issuedAt(Date.from(now))
                .expiration(Date.from(now.plusSeconds(accessTokenSeconds)))
                .signWith(key)
                .compact();
    }

    /**
     * Vérifie signature, expiration, émetteur et type du jeton.
     * Retourne le nom d'utilisateur, ou lève une JwtException.
     */
    public String parseAccessToken(String token) {

        Claims claims = Jwts.parser()
                .verifyWith(key)
                .requireIssuer(ISSUER)
                .build()
                .parseSignedClaims(token)
                .getPayload();

        if (!ACCESS_TYPE.equals(claims.get(TYPE_CLAIM, String.class))) {
            throw new JwtException("Type de jeton invalide.");
        }

        return claims.getSubject();
    }

    public long getAccessTokenSeconds() {
        return accessTokenSeconds;
    }
}

/**

 * après trop d'échecs consécutifs.
 * REQUIRES_NEW : le compteur est enregistré même si la connexion échoue.
 */
@Service

    private final UserRepository userRepository;
    private final int maxFailedAttempts;
    private final Duration lockDuration;

    public LoginAttemptService(
            UserRepository userRepository,
            @Value("${app.security.max-failed-attempts:5}") int maxFailedAttempts,
            @Value("${app.security.lock-minutes:15}") long lockMinutes) {

        this.maxFailedAttempts = maxFailedAttempts;
        this.lockDuration = Duration.ofMinutes(lockMinutes);

    public void recordFailure(String username) {

        userRepository.findByUsernameIgnoreCase(username).ifPresent(user -> {

            userRepository.incrementFailedAttempts(user.getId());

            if (user.getFailedAttempts() + 1 >= maxFailedAttempts) {
                userRepository.lock(user.getId(), Instant.now().plus(lockDuration));
            }
        });
    }

    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void recordSuccess(Long userId) {
        userRepository.registerSuccessfulLogin(userId, Instant.now());
    }

/**
 * Jetons de rafraîchissement : aléatoires, hachés en base, à usage unique (rotation).
 * Réutiliser un jeton déjà consommé révoque toutes les sessions de l'utilisateur
 * (signe probable de vol de jeton).
 */
@Service
public class RefreshTokenService {

    public record Rotation(User user, String refreshToken) {
    }

    private final RefreshTokenRepository repository;
    private final UserRepository userRepository;
    private final long validityDays;
    private final SecureRandom random = new SecureRandom();

    public RefreshTokenService(
            RefreshTokenRepository repository,
            UserRepository userRepository,
            @Value("${app.security.jwt.refresh-token-days:7}") long validityDays) {

        this.repository = repository;
        this.userRepository = userRepository;
        this.validityDays = validityDays;
    }

    @Transactional
    public String issue(Long userId) {

        repository.deleteExpired(Instant.now());

        byte[] bytes = new byte[32];
        random.nextBytes(bytes);

        String raw = Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);

        repository.save(new RefreshToken(
                hash(raw),
                userRepository.getReferenceById(userId),
                Instant.now().plus(validityDays, ChronoUnit.DAYS)
        ));

        return raw;
    }

    @Transactional(noRollbackFor = BadCredentialsException.class)
    public Rotation rotate(String rawToken) {

        RefreshToken token = repository.findByTokenHash(hash(rawToken))
                .orElseThrow(() -> new BadCredentialsException(
                        "Session invalide. Veuillez vous reconnecter."));

        if (token.isRevoked()) {
            repository.revokeAllByUserId(token.getUser().getId());
            throw new BadCredentialsException(
                    "Session invalide. Veuillez vous reconnecter.");
        }

        if (token.getExpiresAt().isBefore(Instant.now())) {
            throw new BadCredentialsException(
                    "Votre session a expiré. Veuillez vous reconnecter.");
        }

        User user = token.getUser();

        if (!user.isEnabled() || user.isLocked()) {
            throw new BadCredentialsException(
                    "Ce compte n’est plus autorisé à se connecter.");
        }

        token.setRevoked(true);

        return new Rotation(user, issue(user.getId()));
    }

    @Transactional
    public void revoke(String rawToken) {
        repository.findByTokenHash(hash(rawToken))
                .ifPresent(token -> token.setRevoked(true));
    }

    @Transactional
    public void revokeAllForUser(Long userId) {
        repository.revokeAllByUserId(userId);
    }

    @Transactional
    public void deleteAllForUser(Long userId) {
        repository.deleteByUserId(userId);
    }

    private static String hash(String value) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            return HexFormat.of().formatHex(
                    digest.digest(value.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 indisponible", e);
        }
    }
}

public final class RestErrors {

    private RestErrors() {
    }

    public static void write(
            HttpServletRequest request,
            HttpServletResponse response,
            int status,
            String error,
            String message,
            String code) throws IOException {

        response.setStatus(status);
        response.setContentType("application/json");
        response.setCharacterEncoding("UTF-8");

        String body = "{"
                + "\"timestamp\":\"" + LocalDateTime.now() + "\","
                + "\"status\":" + status + ","
                + "\"error\":\"" + escape(error) + "\","
                + "\"message\":\"" + escape(message) + "\","
                + "\"path\":\"" + escape(request.getRequestURI()) + "\","
                + "\"code\":\"" + escape(code) + "\""
                + "}";

        response.getWriter().write(body);
    }

    private static String escape(String value) {
        return value == null
                ? ""
                : value.replace("\\", "\\\\")
                        .replace("\"", "\\\"")
                        .replace("\n", " ")
                        .replace("\r", " ");
    }
}

/**
 * CQRS - côté ÉCRITURE.
 * Une commande modifie l'état et ne retourne rien (ou juste l'id créé) :
 * la relecture passe par le côté Query.
 * Optimistic locking : la version envoyée par le client doit être celle en base.
 * Les règles métier se placent dans les hooks, jamais dans le CRUD.
 */
@Transactional
public abstract class BaseCommand<E extends BaseEntity, Q extends VersionedRequest,
        R extends BaseRepository<E>> extends BaseHandler<E, R> {

    protected final EntityMapper<E, Q, ?> mapper;

    protected BaseCommand(final R repository, final EntityMapper<E, Q, ?> mapper) {
        this(repository, mapper, null);
    }

    protected BaseCommand(final R repository, final EntityMapper<E, Q, ?> mapper, final String label) {
        super(repository, label);
        this.mapper = mapper;
    }

    public Long create(Q request) {
        E entity = mapper.toEntity(request);
        prepare(entity);
        validateCreate(entity);
        return repository.save(entity).getId();
    }

    public void update(Long id, Q request) {
        if (request.version() == null) {
            throw new IllegalArgumentException("La version est obligatoire pour une modification.");
        }

        E existing = getOrThrow(id);
        checkVersion(existing, request.version());

        E incoming = mapper.toEntity(request);
        prepare(incoming);
        validateUpdate(existing, incoming);
        copy(existing, incoming);

        repository.saveAndFlush(existing);
    }

    public void delete(Long id) {
        repository.delete(getOrThrow(id));
    }

    /** Refuse l'opération si l'entité a changé depuis la lecture du client. */
    protected void checkVersion(E existing, Long clientVersion) {
        if (clientVersion != null && !clientVersion.equals(existing.getVersion())) {
            throw new ObjectOptimisticLockingFailureException(existing.getClass(), existing.getId());
        }
    }

    // ---------- HOOKS ----------

    protected void prepare(E entity) {
    }

    protected void validateCreate(E entity) {
    }

    protected void validateUpdate(E existing, E incoming) {
    }

    protected abstract void copy(E target, E source);
}

        int currentYear = Year.now().getValue();

        Integer fiscalYear = allBudgets.stream()
                .map(Budget::getFiscalYear)
                .filter(Objects::nonNull)
                .filter(year -> year == currentYear)
                .findFirst()
                .orElseGet(() -> allBudgets.stream()
                        .map(Budget::getFiscalYear)
                        .filter(Objects::nonNull)
                        .max(Integer::compareTo)
                        .orElse(currentYear));

        // 3. Sélection des budgets de l'exercice
        List<Budget> yearBudgets = allBudgets.stream()
                .filter(b -> Objects.equals(b.getFiscalYear(), fiscalYear))
                .toList();

        // Une seule devise est utilisée pour les totaux affichés.
        String currency = yearBudgets.stream()
                .map(Budget::getCurrency)
                .filter(Objects::nonNull)
                .findFirst()
                .orElse("USD");

        List<Budget> budgets = yearBudgets.stream()
                .filter(b -> currency.equals(b.getCurrency()))
                .toList();

        long totalBudgets = budgets.size();

        // 4. Calcul du budget total
        BigDecimal totalBudgetAmount = budgets.stream()
                .map(Budget::getTotalAmount)
                .filter(Objects::nonNull)
                .reduce(BigDecimal.ZERO, BigDecimal::add);

        // 5. Identification des budgets concernés
        Set<Long> selectedBudgetIds = budgets.stream()
                .map(Budget::getId)
                .filter(Objects::nonNull)
                .collect(Collectors.toSet());

        // 6. Filtrage des dépenses liées à ces budgets
        List<Expense> expenses = allExpenses.stream()
                .filter(expense -> expense.getBudget() != null)
                .filter(expense ->
                        selectedBudgetIds.contains(expense.getBudget().getId()))
                .toList();

        // 7. Comptage des dépenses par statut
        long pending = expenses.stream()
                .filter(e -> e.getStatus() == ExpenseStatus.PENDING)
                .count();

        long approved = expenses.stream()
                .filter(e -> e.getStatus() == ExpenseStatus.APPROVED)
                .count();

        long rejected = expenses.stream()
                .filter(e -> e.getStatus() == ExpenseStatus.REJECTED)
                .count();

        ExpenseStatusDto expenseStatus =
                new ExpenseStatusDto(pending, approved, rejected);

        // 8. Calcul du montant réellement dépensé
        // Seules les dépenses approuvées sont prises en compte.
        BigDecimal totalSpentAmount = expenses.stream()
                .filter(e -> e.getStatus() == ExpenseStatus.APPROVED)
                .map(Expense::getAmount)
                .filter(Objects::nonNull)
                .reduce(BigDecimal.ZERO, BigDecimal::add);

        // 9. Calcul de l'exécution budgétaire globale
        double overallExecutionRate =
                calculatePercentage(totalSpentAmount, totalBudgetAmount);

        // 10. Regroupement des budgets par département
        Map<Long, BigDecimal> budgetMap = budgets.stream()
                .filter(b -> b.getDepartment() != null)
                .collect(Collectors.groupingBy(
                        b -> b.getDepartment().getId(),
                        Collectors.reducing(
                                BigDecimal.ZERO,
                                Budget::getTotalAmount,
                                BigDecimal::add
                        )
                ));

        // 11. Regroupement des dépenses approuvées par département
        Map<Long, BigDecimal> spentMap = expenses.stream()
                .filter(e -> e.getStatus() == ExpenseStatus.APPROVED)
                .filter(e -> e.getBudget() != null)
                .filter(e -> e.getBudget().getDepartment() != null)
                .collect(Collectors.groupingBy(
                        e -> e.getBudget().getDepartment().getId(),
                        Collectors.reducing(
                                BigDecimal.ZERO,
                                Expense::getAmount,
                                BigDecimal::add
                        )
                ));

        // 12. Construction des indicateurs par département
        List<DepartmentExecutionDto> departmentExecutions =
                departments.stream()
                        .map(department -> {

                            BigDecimal allocated = budgetMap.getOrDefault(
                                    department.getId(),
                                    BigDecimal.ZERO
                            );

                            BigDecimal spent = spentMap.getOrDefault(
                                    department.getId(),
                                    BigDecimal.ZERO
                            );

                            double percentage =
                                    calculatePercentage(spent, allocated);

                            return new DepartmentExecutionDto(
                                    department.getId(),
                                    department.getName(),
                                    allocated,
                                    spent,
                                    percentage
                            );
                        })
                        .toList();

        // 13. Activités récentes provenant de Hibernate Envers
        List<RecentActivityDto> recentActivities =
                fetchRecentActivities();

        // 14. Nombre réel d'opérations auditées
        long totalOperations = countTotalOperations();

        // 15. Retour du tableau de bord
        return new DashboardSummaryResponse(
                totalDepartments,
                totalBudgets,
                totalBudgetAmount,
                currency,
                fiscalYear,
                totalSpentAmount,
                overallExecutionRate,
                totalOperations,
                departmentExecutions,
                recentActivities,
                expenseStatus
        );
    }

    /**
     * Calcule un pourcentage en évitant la division par zéro.
     */
    private double calculatePercentage(
            BigDecimal amount,
            BigDecimal total) {

        if (amount == null || total == null
                || total.compareTo(BigDecimal.ZERO) <= 0) {
            return 0.0;
        }

        double percentage = amount
                .divide(total, 4, RoundingMode.HALF_UP)
                .doubleValue() * 100.0;

        return Math.round(percentage * 10.0) / 10.0;
    }

    /**

     * Les départements, budgets et dépenses sont inclus.
     */
    private List<RecentActivityDto> fetchRecentActivities() {

        List<RecentActivityDto> activities = new ArrayList<>();

        String sql = """
            SELECT r.rev, r.revtstmp, r.username,
                   'DEPARTMENT' AS entity,
                   d.revtype, d.name AS entity_name, d.id AS entity_id
            FROM revinfo r
            JOIN departments_aud d ON r.rev = d.rev

                   'BUDGET' AS entity,
                   b.revtype, b.name AS entity_name, b.id AS entity_id
            FROM revinfo r
            JOIN budgets_aud b ON r.rev = b.rev

            UNION ALL

            SELECT r.rev, r.revtstmp, r.username,
                   'EXPENSE' AS entity,
                   e.revtype, e.description AS entity_name,
                   e.id AS entity_id
            FROM revinfo r
            JOIN expenses_aud e ON r.rev = e.rev

            ORDER BY revtstmp DESC, rev DESC
            LIMIT 6
            """;

        try {
            Query query = entityManager.createNativeQuery(sql);

            @SuppressWarnings("unchecked")
            List<Object[]> rows = query.getResultList();

            for (Object[] row : rows) {

                Long revision = ((Number) row[0]).longValue();
                Long timestamp = ((Number) row[1]).longValue();

                String username = (String) row[2];
                String entity = (String) row[3];

                int revisionType = ((Number) row[4]).intValue();

                String entityName = (String) row[5];
                Long entityId = ((Number) row[6]).longValue();

                String title;
                String icon;
                String color;

                String entityLabel = switch (entity) {
                    case "DEPARTMENT" -> "Département";
                    case "BUDGET" -> "Budget";
                    case "EXPENSE" -> "Dépense";
                    default -> "Opération";
                };

                switch (revisionType) {
                    case 0 -> {
                        title = "Création : " + entityLabel;
                        icon = "add_circle";
                        color = "green";
                    }
                    case 1 -> {
                        title = "Modification : " + entityLabel;
                        icon = "edit";
                        color = "orange";
                    }
                    case 2 -> {
                        title = "Suppression : " + entityLabel;
                        icon = "delete";
                        color = "red";
                    }
                    default -> {
                        title = "Opération : " + entityLabel;
                        icon = "history";
                        color = "blue";
                    }
                }

                String user = username != null && !username.isBlank()
                        ? username
                        : "Utilisateur inconnu";

                String description = entityName != null
                        ? entityName
                        : entityLabel + " #" + entityId;

                Instant instant = Instant.ofEpochMilli(timestamp);

                activities.add(new RecentActivityDto(

                        title,
                        description,
                        user,
                        formatRelativeTime(timestamp),
                        icon,

                ));

            );

    }

    /**
     * Compte les opérations enregistrées dans les tables d'audit.
     * Une ligne d'une table *_aud représente une opération sur une entité.
     */
    private long countTotalOperations() {

        String sql = """
            SELECT
                (SELECT COUNT(*) FROM departments_aud)
                +
                (SELECT COUNT(*) FROM budgets_aud)
                +
                (SELECT COUNT(*) FROM expenses_aud)
            """;

        try {
            Query query = entityManager.createNativeQuery(sql);

            return ((Number) query.getSingleResult()).longValue();

        } catch (Exception e) {
            log.error("Impossible de compter les opérations auditées", e);
            return 0;
        }
    }

    /**
     * Convertit une date en temps relatif.
     */
    private String formatRelativeTime(Long epochMilli) {

        if (epochMilli == null) {
            return "Récemment";
        }

        Instant instant = Instant.ofEpochMilli(epochMilli);

        Duration duration = Duration.between(
                instant,
                Instant.now()
        );

        long minutes = duration.toMinutes();

        if (minutes < 1) {
            return "À l'instant";
        }

        if (minutes < 60) {
            return "Il y a " + minutes + " min";
        }

        long hours = duration.toHours();

        if (hours < 24) {
            return "Il y a " + hours + " h";
        }

        long days = duration.toDays();

        if (days < 7) {
            return "Il y a " + days + " j";
        }

        DateTimeFormatter formatter =
                DateTimeFormatter.ofPattern("dd/MM/yyyy HH:mm")
                        .withZone(ZoneId.systemDefault());

        return formatter.format(instant);
    }

    protected void prepare(Expense expense) {
        Long budgetId = expense.getBudget().getId();
        expense.setBudget(budgetRepository.findById(budgetId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Budget introuvable avec l'identifiant : " + budgetId)));
    }

    @Override
    protected void validateUpdate(Expense existing, Expense incoming) {
        if (existing.getStatus() == ExpenseStatus.REJECTED) {
            throw new IllegalStateException("Une dépense rejetée ne peut pas être modifiée.");
        }

            validateBudgetCapacity(incoming.getBudget(), incoming.getAmount(), existing.getId());
        }

        target.setDescription(source.getDescription());
        target.setAmount(source.getAmount());
        target.setExpenseDate(source.getExpenseDate());
        target.setBudget(source.getBudget());
    }

    // ---------- COMMANDES MÉTIER ----------

    public void approve(Long id, Long version) {
        Expense expense = getOrThrow(id);
        checkVersion(expense, version);

        if (expense.getStatus() != ExpenseStatus.PENDING) {
            throw new IllegalStateException("Seules les dépenses en attente peuvent être approuvées.");
        }

        validateBudgetCapacity(expense.getBudget(), expense.getAmount(), null);

        expense.setStatus(ExpenseStatus.APPROVED);
        expense.setRejectionReason(null);
        repository.saveAndFlush(expense);
    }

    public void reject(Long id, String reason, Long version) {
        Expense expense = getOrThrow(id);
        checkVersion(expense, version);

        if (expense.getStatus() != ExpenseStatus.PENDING) {
            throw new IllegalStateException("Seules les dépenses en attente peuvent être rejetées.");
        }
        if (reason == null || reason.isBlank()) {
            throw new IllegalArgumentException("Le motif du rejet est obligatoire.");
        }

        expense.setStatus(ExpenseStatus.REJECTED);
        expense.setRejectionReason(reason.trim());
        repository.saveAndFlush(expense);
    }

    // ---------- PRIVÉ ----------

    private void validateBudgetCapacity(Budget budget, BigDecimal newAmount, Long excludedExpenseId) {
        BigDecimal approvedAmount = repository
                .findByBudgetIdAndStatus(budget.getId(), ExpenseStatus.APPROVED)
                .stream()
                .filter(e -> excludedExpenseId == null || !e.getId().equals(excludedExpenseId))
                .map(Expense::getAmount)
                .reduce(BigDecimal.ZERO, BigDecimal::add);

        if (approvedAmount.add(newAmount).compareTo(budget.getTotalAmount()) > 0) {
            throw new IllegalArgumentException(
                    "Budget insuffisant. Montant disponible : "
                            + budget.getTotalAmount().subtract(approvedAmount));
        }
    }

    @Override
    protected void validateUpdate(User existing, User incoming) {

        if (!existing.getUsername().equalsIgnoreCase(incoming.getUsername())) {
            throw new IllegalArgumentException(
                    "Le nom d’utilisateur ne peut pas être modifié.");
        }

        if (incoming.getEmail() != null) {
            repository.findByEmailIgnoreCase(incoming.getEmail())
                    .filter(found -> !found.getId().equals(existing.getId()))
                    .ifPresent(found -> {
                        throw new DuplicateResourceException(
                                "Cette adresse e-mail est déjà utilisée.");
                    });
        }

        boolean wasAdmin = hasAdminRole(existing);
        boolean staysAdmin = hasAdminRole(incoming) && incoming.isEnabled();

        if (CurrentUser.isUsername(existing.getUsername())) {

            if (!incoming.isEnabled()) {
                throw new IllegalStateException(
                        "Vous ne pouvez pas désactiver votre propre compte.");
            }

            if (wasAdmin && !hasAdminRole(incoming)) {
                throw new IllegalStateException(
                        "Vous ne pouvez pas retirer votre propre rôle d’administrateur.");
            }
        }

        if (wasAdmin && existing.isEnabled() && !staysAdmin) {
            ensureAnotherAdmin(existing.getId());
        }
    }

    @Override
    protected void copy(User target, User source) {
        target.setFullName(source.getFullName());
        target.setEmail(source.getEmail());
        target.setEnabled(source.isEnabled());
        target.getRoles().clear();
        target.getRoles().addAll(source.getRoles());
    }

    // ---------- COMMANDES ----------

    @Override
    public void update(Long id, UserRequest request) {
        super.update(id, request);

        // Compte désactivé : toutes ses sessions sont coupées immédiatement.
        if (Boolean.FALSE.equals(request.enabled())) {
            refreshTokenService.revokeAllForUser(id);
        }
    }

    @Override
    public void delete(Long id) {

        User user = getOrThrow(id);

        if (CurrentUser.isUsername(user.getUsername())) {
            throw new IllegalStateException("Vous ne pouvez pas supprimer votre propre compte.");
        }

        if (hasAdminRole(user) && user.isEnabled()) {
            ensureAnotherAdmin(id);
        }

        refreshTokenService.deleteAllForUser(id);
        repository.delete(user);
    }

    // Réinitialisation par un administrateur : mot de passe provisoire.
    public void resetPassword(Long id, String newPassword) {

        User user = getOrThrow(id);

        user.setPassword(passwordEncoder.encode(newPassword));
        user.setMustChangePassword(true);
        user.setFailedAttempts(0);
        user.setLockedUntil(null);

        repository.saveAndFlush(user);

        refreshTokenService.revokeAllForUser(id);
    }

    // ---------- PRIVÉ ----------

    private boolean hasAdminRole(User user) {
        return user.getRoles().stream()
                .anyMatch(role -> RoleNames.ADMIN.equals(role.getName()));
    }

    // Garantit qu'il reste toujours au moins un administrateur actif.
    private void ensureAnotherAdmin(Long excludedUserId) {
        if (repository.countOtherEnabledWithRole(RoleNames.ADMIN, excludedUserId) == 0) {
            throw new IllegalStateException(
                    "Opération refusée : il doit rester au moins un administrateur actif.");
        }
    }
}
